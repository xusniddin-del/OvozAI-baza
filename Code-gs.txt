function doGet(e) {
  if (e && e.parameter && e.parameter.mcp != null) return json_({ jsonrpc: '2.0', id: null, error: { code: -32000, message: 'Gapir AI MCP: faqat POST so\'rovlar qabul qilinadi' } });
  return json_({ ok: true, service: 'Gapir AI', time: new Date().toISOString() });
}

function doPost(e) {
  if (e && e.parameter && e.parameter.mcp != null) return mcp_(e);   // Claude konnektori
  try {
    const req = JSON.parse((e && e.postData && e.postData.contents) || '{}');
    if (req.update_id != null) {   // Telegram: HtmlService 302 emas, to'g'ridan-to'g'ri 200 qaytaradi
      try { tgUpdate_(req); } catch (te) { console.error(te); }
      return HtmlService.createHtmlOutput('ok');
    }
    try { ensureTextFmt_(); ensureHeaders_(); } catch (fe) { console.error(fe); }
    switch (req.action) {
      case 'cfg':      return json_({ ok: true, bot: botName_(), prices: set_().prices, discount: set_().discount, limits: set_().plans, pay: !!tcKey_() });
      case 'pay_create': return json_(payCreate_(req));
      case 'pay_check':  return json_(payCheck_(req));
      case 'mcp_key':    return json_(mcpUserKey_(req));
      case 'image':      return json_(image_(req));
      case 'ticket':     return json_(ticket_(req));
      case 'svc_refund': return json_(svcRefundTicket_(req));
      case 'ticket_use': return json_(ticketUse_(req));
      case 'site_publish': return json_(sitePublish_(req));
      case 'app_build':  return json_(appBuild_(req));
      case 'app_status': return json_(appStatus_(req));
      case 'vcheck':   return json_(vcheck_(req));
      case 'reset':    return json_(reset_(req));
      case 'chat':     return json_(chat_(req));
      case 'register': return json_(register_(req));
      case 'login':    return json_(login_(req));
      case 'me':       return json_(me_(req));
      case 'logout':   return json_(logout_(req));
      case 'tts':      return json_(tts_(req));
      case 'stt':      return json_(stt_(req));
      case 'translate':return json_(translate_(req));
      case 'admin':    return json_(admin_(req));
      default:         return json_({ ok: false, error: "Noma'lum amal" });
    }
  } catch (err) {
    console.error(err && err.stack || err);
    return json_({ ok: false, error: 'Server xatosi. Birozdan keyin urinib ko\'ring.' });
  }
}

function json_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

/** Raqamni +998XXXXXXXXX ko'rinishiga keltiradi. Noto'g'ri bo'lsa '' qaytaradi. */
function normPhone_(raw) {
  let d = String(raw || '').replace(/\D/g, '');
  if (d.length === 9) d = '998' + d;
  if (d.length === 12 && d.indexOf('998') === 0) return '+' + d;
  return '';
}

function register_(req) {
  const name = String(req.name || '').trim().slice(0, 60);
  let phone = normPhone_(req.phone), tgId = '';
  const password = String(req.password || '');
  if (name.length < 2) return { ok: false, error: 'Ismingizni kiriting' };
  if (botToken_()) {
    const v = vget_(req.vcode);
    if (!v || v.purpose !== 'r' || v.st !== 'ok') return { ok: false, error: "Avval raqamingizni Telegram bot orqali tasdiqlang" };
    phone = v.phone; tgId = v.tg;
  }
  if (!phone) return { ok: false, error: "Telefon raqam noto'g'ri (masalan: +998 90 123 45 67)" };
  if (password.length < CFG.MIN_PASSWORD) return { ok: false, error: 'Parol kamida ' + CFG.MIN_PASSWORD + ' belgi bo\'lsin' };

  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    if (findPhoneRow_(phone)) return { ok: false, error: "Bu raqam bilan ro'yxatdan o'tilgan. Kirish bo'limidan kiring." };
    const salt = Utilities.getUuid();
    const rc = String(req.ref || '').toUpperCase().replace(/[^A-Z0-9]/g, '').slice(0, 12);
    const referredBy = rc && findRow_('Users', USER_HEADERS, 'ref', rc) ? rc : '';
    const user = {
      id: Utilities.getUuid(), name: name, phone: phone, salt: salt,
      hash: hashPassword_(password, salt), plan: 'free', usedToday: 0,
      day: today_(), createdAt: new Date().toISOString(), status: 'active', lastLogin: new Date().toISOString(),
      ref: makeRef_(), referredBy: referredBy, tgId: tgId
    };
    const ush = sheet_('Users', USER_HEADERS);
    ush.appendRow(USER_HEADERS.map(function (h) { return user[h] == null ? "" : user[h]; }));
    ush.getRange(ush.getLastRow(), 1, 1, USER_HEADERS.length).setNumberFormat('@')
       .setValues([USER_HEADERS.map(function (h) { return user[h] == null ? "" : String(user[h]); })]);
    if (req.vcode) CacheService.getScriptCache().remove('v_' + req.vcode);
    if (tgId) tgSend_(tgId, '🎉 <b>' + esc_(name) + "</b>, ro'yxatdan muvaffaqiyatli o'tdingiz!\nKunlik bepul limit: <b>" + limitFor_(user) + '</b> belgi.', siteBtn_());
    return { ok: true, token: createSession_(user.id), user: publicUser_(user) };
  } finally {
    lock.releaseLock();
  }
}

function login_(req) {
  const phone = normPhone_(req.phone);
  const password = String(req.password || '');
  if (!phone) return { ok: false, error: "Telefon raqam noto'g'ri (masalan: +998 90 123 45 67)" };
  const cache = CacheService.getScriptCache();
  const failKey = 'fail_' + hex_(Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, phone)).slice(0, 24);
  if (Number(cache.get(failKey) || 0) >= CFG.MAX_FAILS) {
    return { ok: false, error: "Juda ko'p urinish. 15 daqiqadan keyin qayta urinib ko'ring." };
  }
  const row = findPhoneRow_(phone);
  const user = row ? readUser_(row) : null;
  if (user) ensureRef_(row, user);
  if (!user || user.hash !== hashPassword_(password, user.salt)) {
    cache.put(failKey, String(Number(cache.get(failKey) || 0) + 1), 900);
    return { ok: false, error: "Telefon raqam yoki parol noto'g'ri" };
  }
  if (user.status !== 'active') return { ok: false, error: 'Akkaunt bloklangan' };
  cache.remove(failKey);
  sheet_('Users', USER_HEADERS).getRange(row, USER_HEADERS.indexOf('lastLogin') + 1).setValue(new Date().toISOString());
  return { ok: true, token: createSession_(user.id), user: publicUser_(user) };
}

function me_(req) {
  const auth = auth_(req.token);
  if (!auth) return { ok: false, error: 'auth' };
  ensureRef_(auth.row, auth.user);
  try { if (payPendingFor_(auth.user.id)) auth.user = readUser_(auth.row); } catch (e) { console.error(e); }
  return { ok: true, user: publicUser_(auth.user) };
}

function logout_(req) {
  const row = findRow_('Sessions', SESSION_HEADERS, 'tokenHash', tokenHash_(req.token || ''));
  if (row) sheet_('Sessions', SESSION_HEADERS).deleteRow(row);
  return { ok: true };
}

/* ============ Matnni ovozga aylantirish ============ */

function tts_(req) {
  const auth = auth_(req.token);
  if (!auth) return { ok: false, error: 'auth' };

  const text = String(req.text || '').trim();
  const voice = VOICES.indexOf(req.voice) >= 0 ? req.voice : 'Kore';
  const style = String(req.style || '').trim().slice(0, 160);
  if (!text) return { ok: false, error: 'Matn kiriting' };
  if (text.length > CFG.MAX_CHARS_PER_REQUEST) return { ok: false, error: 'Bir so\'rovga ' + CFG.MAX_CHARS_PER_REQUEST + ' belgigacha' };

  // Limitni oldindan band qilamiz (kvotani aylanib o'tishga yo'l qo'ymaslik uchun)
  const reserve = reserveQuota_(auth.row, text.length);
  if (!reserve.ok) return reserve;

  const key = PropertiesService.getScriptProperties().getProperty('GEMINI_API_KEY');
  if (!key) { refundQuota_(auth.row, text.length); return { ok: false, error: 'Server sozlanmagan (API kalit yo\'q)' }; }

  const result = callGemini_(text, voice, style, key);
  if (!result.ok) {
    refundQuota_(auth.row, text.length);
    log_(auth.user.id, text.length, result.model || '', false, result.error);
    return { ok: false, error: result.userError || 'Ovoz yaratilmadi. Qayta urinib ko\'ring.' };
  }
  log_(auth.user.id, text.length, result.model, true, '');
  return { ok: true, audio: result.audio, mime: result.mime, used: reserve.used, limit: reserve.limit };
}

function attempts_() {
  // Hujjatlardagi ikki xil so'rov shakli: yangi (3.8) va eski (prebuiltVoiceConfig)
  return [
    { model: CFG.MODEL, shape: 'new' },
    { model: CFG.MODEL, shape: 'old' },
    { model: CFG.FALLBACK_MODEL, shape: 'new' },
    { model: CFG.FALLBACK_MODEL, shape: 'old' },
    { model: 'gemini-2.5-flash-preview-tts', shape: 'old' }
  ];
}

function buildBody_(shape, text, voice, style) {
  if (shape === 'new') {
    const part = { text: text };
    if (style) part.speech_metadata = { style: style };
    return {
      contents: [{ role: 'user', parts: [part] }],
      generationConfig: { responseModalities: ['AUDIO'], speechConfig: { voiceConfig: { voice: voice } } }
    };
  }
  const prompt = style ? 'Say ' + style + ': ' + text : text;
  return {
    contents: [{ parts: [{ text: prompt }] }],
    generationConfig: {
      responseModalities: ['AUDIO'],
      speechConfig: { voiceConfig: { prebuiltVoiceConfig: { voiceName: voice } } }
    }
  };
}

function callGemini_(text, voice, style, key) {
  const props = PropertiesService.getScriptProperties();
  const known = props.getProperty('WORKING_ATTEMPT'); // masalan "gemini-3.8-flash-tts|new"
  let list = attempts_();
  if (known) {
    list.sort(function (a, b) { return (b.model + '|' + b.shape === known) - (a.model + '|' + a.shape === known); });
  }
  let lastErr = '', userError = '', lastModel = '';
  const busy = {};   // limiti tugagan modellar — ularni qayta sinamaymiz
  for (let i = 0; i < list.length; i++) {
    const a = list[i];
    if (busy[a.model]) continue;
    lastModel = a.model;
    const resp = UrlFetchApp.fetch('https://generativelanguage.googleapis.com/v1beta/models/' + a.model + ':generateContent', {
      method: 'post',
      contentType: 'application/json',
      headers: { 'x-goog-api-key': key },
      payload: JSON.stringify(buildBody_(a.shape, text, voice, style)),
      muteHttpExceptions: true
    });
    const code = resp.getResponseCode();
    const raw = resp.getContentText();
    if (code === 200) {
      let data;
      try { data = JSON.parse(raw); } catch (e) { lastErr = 'JSON xato'; continue; }
      const parts = (((data.candidates || [])[0] || {}).content || {}).parts || [];
      for (let p = 0; p < parts.length; p++) {
        const inl = parts[p].inlineData || parts[p].inline_data;
        if (inl && inl.data) {
          props.setProperty('WORKING_ATTEMPT', a.model + '|' + a.shape);
          return { ok: true, audio: inl.data, mime: inl.mimeType || inl.mime_type || 'audio/L16;rate=24000', model: a.model };
        }
      }
      lastErr = 'Audio qaytmadi: ' + raw.slice(0, 200);
      continue;
    }
    lastErr = code + ': ' + raw.slice(0, 300);
    if (code === 429 || code === 503) {   // shu modelning limiti tugagan yoki band — boshqa modelni sinaymiz
      busy[a.model] = true;
      userError = /per.?day|PerDay|daily/i.test(raw)
        ? "Bugungi bepul Google limiti tugadi. Ertaga qayta urinib ko'ring yoki admin billing'ni yoqsin."
        : "Hozir so'rovlar ko'p. Bir daqiqadan keyin urinib ko'ring.";
      continue;
    }
    if (code === 401 || code === 403) { userError = 'Server API kaliti bilan muammo. Admin bilan bog\'laning.'; break; }
    // 400/404 — shakl yoki model mos kelmadi: keyingi variantga o'tamiz
  }
  return { ok: false, error: lastErr, userError: userError, model: lastModel };
}

/* ============ Nutqdan matnga va tarjima (Gemini matn modeli) ============ */

function stt_(req) {
  const auth = auth_(req.token);
  if (!auth) return { ok: false, error: 'auth' };
  const audio = String(req.audio || '');
  const mime = String(req.mime || 'audio/mpeg').split(';')[0];
  if (!audio) return { ok: false, error: 'Audio fayl yuborilmadi' };
  if (audio.length > CFG.MAX_AUDIO_B64) return { ok: false, error: 'Fayl juda katta (13 MB gacha)' };
  if (!/^(audio|video)\//.test(mime)) return { ok: false, error: "Faqat audio yoki video fayl" };
  const user = readUser_(auth.row), limit = limitFor_(user);
  const used = user.day === today_() ? Number(user.usedToday) || 0 : 0;
  if (used >= limit) return { ok: false, error: 'limit', message: 'Bugungi limit tugadi (' + used + '/' + limit + ').', used: used, limit: limit };
  const lang = String(req.lang || '').slice(0, 30);
  const prompt = 'Transcribe this audio verbatim' + (lang ? ' (language: ' + lang + ')' : '') +
    '. Output only the transcript text in the original language, with punctuation and paragraph breaks. ' +
    'If there are several speakers, prefix lines with "Spiker 1:", "Spiker 2:". Do not add any comments.';
  const r = callText_([{ inline_data: { mime_type: mime, data: audio } }, { text: prompt }]);
  if (!r.ok) { log_(auth.user.id, 0, r.model || '', false, 'stt ' + r.error); return { ok: false, error: r.userError || "Matnga o'girilmadi. Qayta urinib ko'ring." }; }
  const charged = chargeQuota_(auth.row, r.text.length);
  log_(auth.user.id, r.text.length, r.model, true, 'stt');
  return { ok: true, text: r.text, used: charged.used, limit: charged.limit };
}

function translate_(req) {
  const auth = auth_(req.token);
  if (!auth) return { ok: false, error: 'auth' };
  const text = String(req.text || '').trim();
  const lang = String(req.lang || '').slice(0, 30);
  if (!text || !lang) return { ok: false, error: 'Matn va til kerak' };
  if (text.length > CFG.MAX_TRANSLATE) return { ok: false, error: 'Tarjima uchun ' + CFG.MAX_TRANSLATE + ' belgigacha' };
  const reserve = reserveQuota_(auth.row, text.length);
  if (!reserve.ok) return reserve;
  const r = callText_([{ text: 'Translate the following text into ' + lang + '. Keep the meaning, tone and line breaks. Output only the translation.\n\n' + text }]);
  if (!r.ok) { refundQuota_(auth.row, text.length); return { ok: false, error: r.userError || 'Tarjima qilinmadi' }; }
  log_(auth.user.id, text.length, r.model, true, 'translate');
  return { ok: true, text: r.text, used: reserve.used, limit: reserve.limit };
}

function callText_(parts, sys, history, opt) {
  opt = opt || {};
  const P = PropertiesService.getScriptProperties();
  // Yordamchi (chat) uchun alohida kalit bo'lsa — o'shani ishlatadi, aks holda asosiy kalitni
  const key = (opt.chat && String(P.getProperty('CHAT_API_KEY') || '').trim()) || P.getProperty('GEMINI_API_KEY');
  if (!key) return { ok: false, error: 'no key', userError: "Server sozlanmagan (API kalit yo'q)" };
  const cache = CacheService.getScriptCache(), tag = opt.chat ? 'c' : 'm';
  let lastErr = '', userError = '', model = '';
  // Yaqinda band bo'lgan modellarni oxiriga suradi — har safar ularni kutib o'tirmaslik uchun
  const LIST = opt.chat ? CFG.CHAT_MODELS : CFG.TEXT_MODELS, tries = [];
  const models = LIST.filter(function (m) { return !cache.get('busy' + tag + m); })
    .concat(LIST.filter(function (m) { return cache.get('busy' + tag + m); }));
  for (let i = 0; i < models.length; i++) {
    model = models[i];
    const gen = sys ? { maxOutputTokens: opt.chat ? 450 : 600 } : {};
    if (sys && /^gemini-2\./.test(model)) gen.temperature = 0.7;   // Gemini 3.x da temperature eskirgan
    // Chatda "o'ylash"ni o'chiradi (lite modellarda u odatda minimal) — javob tezroq keladi
    const fast = opt.chat && !/lite/.test(model) && !cache.get('nofast' + model);
    if (fast) gen.thinkingConfig = /^gemini-2\./.test(model) ? { thinkingBudget: 0 } : { thinkingLevel: 'low' };
    const body = { contents: (history || []).concat([{ role: 'user', parts: parts }]) };
    if (sys) body.systemInstruction = { parts: [{ text: sys }] };
    if (Object.keys(gen).length) body.generationConfig = gen;
    const resp = UrlFetchApp.fetch('https://generativelanguage.googleapis.com/v1beta/models/' + model + ':generateContent', {
      method: 'post', contentType: 'application/json', headers: { 'x-goog-api-key': key },
      payload: JSON.stringify(body), muteHttpExceptions: true
    });
    const code = resp.getResponseCode(), raw = resp.getContentText();
    if (code === 200) {
      try {
        const ps = ((((JSON.parse(raw).candidates || [])[0] || {}).content || {}).parts) || [];
        const text = ps.filter(function (p) { return !p.thought; }).map(function (p) { return p.text || ''; }).join('').trim();
        if (text) return { ok: true, text: text, model: model };
        lastErr = "Bo'sh javob";
      } catch (e) { lastErr = 'JSON xato'; }
      continue;
    }
    lastErr = code + ': ' + raw.slice(0, 300);
    tries.push(model.replace('gemini-', '') + ' ' + code);
    if (code === 400 && fast && /thinking/i.test(raw)) {    // model bu sozlamani bilmasa — usiz qayta urinadi
      cache.put('nofast' + model, '1', 21600); i--; continue;
    }
    if (code === 404) { cache.put('busy' + tag + model, '1', 21600); continue; }   // model bu kalitga yopiq — 6 soat chetlab o'tadi
    if (code === 429 || code === 503) {
      cache.put('busy' + tag + model, '1', code === 429 ? 300 : 60);
      userError = "Hozir so'rovlar ko'p. Bir daqiqadan keyin urinib ko'ring."; continue;
    }
    if (code === 401 || code === 403) { userError = "Server API kaliti bilan muammo. Admin bilan bog'laning."; break; }
  }
  return { ok: false, error: (tries.length ? '[' + tries.join(', ') + '] ' : '') + lastErr, userError: userError, model: model };
}

/** Bajarilgan ish uchun kvotani yechadi (natija uzunligi oldindan noma'lum bo'lganda) */
function chargeQuota_(row, chars) {
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const user = readUser_(row), limit = limitFor_(user);
    const used = (user.day === today_() ? Number(user.usedToday) || 0 : 0) + chars;
    sheet_('Users', USER_HEADERS).getRange(row, USER_HEADERS.indexOf('usedToday') + 1, 1, 2).setValues([[used, today_()]]);
    return { used: used, limit: limit };
  } finally { lock.releaseLock(); }
}

/* ============ Kvota ============ */

function reserveQuota_(row, chars) {
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const user = readUser_(row);
    const limit = limitFor_(user);
    let used = user.day === today_() ? Number(user.usedToday) || 0 : 0;
    if (used + chars > limit) {
      return { ok: false, error: 'limit', message: 'Bugungi limit tugadi (' + used + '/' + limit + '). Ertaga yangilanadi yoki tarifni oshiring.', used: used, limit: limit };
    }
    used += chars;
    const sh = sheet_('Users', USER_HEADERS);
    sh.getRange(row, USER_HEADERS.indexOf('usedToday') + 1, 1, 2).setValues([[used, today_()]]);
    return { ok: true, used: used, limit: limit };
  } finally {
    lock.releaseLock();
  }
}

function refundQuota_(row, chars) {
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const user = readUser_(row);
    if (user.day !== today_()) return;
    const used = Math.max(0, (Number(user.usedToday) || 0) - chars);
    sheet_('Users', USER_HEADERS).getRange(row, USER_HEADERS.indexOf('usedToday') + 1).setValue(used);
  } finally {
    lock.releaseLock();
  }
}

}

function hashPassword_(password, salt) {
  let h = salt + ':' + password;
  for (let i = 0; i < CFG.HASH_ROUNDS; i++) {
    h = hex_(Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, h + salt));
  }
  return h;
}

function tokenHash_(token) {
  return hex_(Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, String(token)));
}

function hex_(bytes) {
  return bytes.map(function (b) { return ('0' + (b < 0 ? b + 256 : b).toString(16)).slice(-2); }).join('');
}

function adminStats_() {
  const users = allUsers_(), today = today_(), S = set_();
  const days = [];
  for (let i = 13; i >= 0; i--) days.push(Utilities.formatDate(new Date(Date.now() - i * 86400000), CFG.TZ, 'yyyy-MM-dd'));
  const idx = {}; days.forEach(function (d, i) { idx[d] = i; });
  const reg = days.map(function () { return 0; }), chars = reg.slice(), reqs = reg.slice(), errs = reg.slice();
  const plans = {}; Object.keys(S.plans).forEach(function (p) { plans[p] = 0; });
  let newToday = 0, new7 = 0, active = 0, used = 0, blocked = 0;
  users.forEach(function (u) {
    const cd = dayOf_(u.createdAt);
    if (cd in idx) reg[idx[cd]]++;
    if (cd === today) newToday++;
    if (cd && idx[cd] >= 7) new7++;
    if (u.day === today && Number(u.usedToday) > 0) { active++; used += Number(u.usedToday) || 0; }
    plans[u.plan] = (plans[u.plan] || 0) + 1;
    if (u.status !== 'active') blocked++;
  });
  const kinds = { tts: 0, stt: 0, translate: 0 };
  const lg = sheet_('Log', LOG_HEADERS), last = lg.getLastRow();
  if (last >= 2) {
    const from = Math.max(2, last - 5000);
    lg.getRange(from, 1, last - from + 1, LOG_HEADERS.length).getValues().forEach(function (r) {
      const d = dayOf_(r[0]); if (!(d in idx)) return;
      const i = idx[d], ok = r[4] === true || r[4] === 'TRUE', note = String(r[5] || '');
      reqs[i]++; if (!ok) errs[i]++; else chars[i] += Number(r[2]) || 0;
      if (d === today && note.indexOf('chat') !== 0) { const k = note.indexOf('stt') === 0 ? 'stt' : note.indexOf('translate') === 0 ? 'translate' : 'tts'; kinds[k]++; }
    });
  }
  const top = users.filter(function (u) { return u.day === today && Number(u.usedToday) > 0; })
    .sort(function (a, b) { return b.usedToday - a.usedToday; }).slice(0, 6)
    .map(function (u) { return { name: u.name, phone: u.phone, used: Number(u.usedToday), plan: u.plan }; });
  return { ok: true, stats: { total: users.length, newToday: newToday, new7: new7, active: active, usedToday: used, blocked: blocked,
    plans: plans, days: days, reg: reg, chars: chars, reqs: reqs, errs: errs, kinds: kinds, top: top } };
}

function botName_() {
  const tok = botToken_(); if (!tok) return '';
  const p = PropertiesService.getScriptProperties(), saved = p.getProperty('BOT_USERNAME_FOR');
  if (saved === tok.slice(-12) && p.getProperty('BOT_USERNAME')) return p.getProperty('BOT_USERNAME');
  try {
    const r = JSON.parse(UrlFetchApp.fetch('https://api.telegram.org/bot' + tok + '/getMe', { muteHttpExceptions: true }).getContentText());
    if (r.ok) { p.setProperty('BOT_USERNAME', r.result.username); p.setProperty('BOT_USERNAME_FOR', tok.slice(-12)); return r.result.username; }
  } catch (e) { console.error(e); }
  return '';
}

  } catch (e) { console.error(e); return null; }
}

const PAY_HEADERS = ['bill','userId','plan','months','amount','state','createdAt','paidAt','env','ref'];

function tcKey_() { return String(PropertiesService.getScriptProperties().getProperty('TEZCHECK_KEY') || '').trim(); }

function tcCall_(path, body, idem) {
  const p = PropertiesService.getScriptProperties();
  const headers = { Authorization: 'Bearer ' + tcKey_() };
  const desk = String(p.getProperty('TEZCHECK_DESK') || '').trim();
  if (desk) headers['X-Cash-Desk-Code'] = desk;
  if (idem) headers['Idempotency-Key'] = idem;
  const resp = UrlFetchApp.fetch(CFG.TC_API + path, { method: 'post', contentType: 'application/json', headers: headers,
    payload: JSON.stringify(body || {}), muteHttpExceptions: true });
  let json = {}; try { json = JSON.parse(resp.getContentText()); } catch (e) {}
  return { code: resp.getResponseCode(), json: json };
}

function priceOf_(plan, months) {
  const S = set_(), m = Number(S.prices[plan]) || 0;
  if (!m) return 0;
  // Yillik: chegirmali oylik narx 1 000 so'mga yaxlitlanadi va 12 ga ko'paytiriladi (39 000 × 12)
  return months === 12 ? Math.round(m * (100 - S.discount) / 100 / 1000) * 1000 * 12 : Math.round(m / 1000) * 1000;
}

/** Bitta to'lovni TezCheck'dan tekshiradi; to'langan bo'lsa tarifni yoqadi. */
function payVerify_(row) {
  const sh = sheet_('Payments', PAY_HEADERS);
  const v = sh.getRange(row, 1, 1, PAY_HEADERS.length).getValues()[0], p = {};
  PAY_HEADERS.forEach(function (h, i) { p[h] = v[i]; });
  if (p.state !== 'pending') return p;
  const r = tcCall_('/bills/' + encodeURIComponent(p.bill), {});
  const b = r.json && r.json.data && r.json.data.bill;
  if (!b) return p;
  const col = function (h) { return PAY_HEADERS.indexOf(h) + 1; };
  if (b.paid) {
    const env = String(b.environment || (b.livemode === false ? 'test' : 'live'));
    const uRow = findRow_('Users', USER_HEADERS, 'id', String(p.userId));
    const user = uRow ? readUser_(uRow) : null;
    if (env === 'test' && !(user && isAdmin_(user))) {   // test to'lovi faqat admin uchun tarifni yoqadi
      sh.getRange(row, col('state'), 1, 1).setValue('test');
      sh.getRange(row, col('env')).setValue(env);
      p.state = 'test'; return p;
    }
    if (uRow) activatePlan_(uRow, String(p.plan), Number(p.months));
    sh.getRange(row, col('state')).setValue('paid');
    sh.getRange(row, col('paidAt')).setValue(new Date().toISOString());
    sh.getRange(row, col('env')).setValue(env);
    p.state = 'paid';
    if (user && user.tgId) tgSend_(user.tgId, '✅ To\'lov qabul qilindi! <b>' + esc_(String(p.plan).toUpperCase()) + '</b> tarifi yoqildi. Rahmat! 🎉', siteBtn_());
  } else if (['expired', 'revoked', 'exhausted'].indexOf(String(b.state)) >= 0) {
    sh.getRange(row, col('state')).setValue(String(b.state)); p.state = String(b.state);
  }
  return p;
}

function activatePlan_(uRow, plan, months) {
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    const sh = sheet_('Users', USER_HEADERS), col = function (h) { return USER_HEADERS.indexOf(h) + 1; };
    const u = readUser_(uRow);
    let from = Date.now();
    const cur = u.planUntil ? new Date(u.planUntil).getTime() : 0;
    if (u.plan === plan && cur > from) from = cur;              // shu tarif davom etsa — muddatga qo'shiladi
    const until = new Date(from); until.setMonth(until.getMonth() + (months || 1));
    sh.getRange(uRow, col('plan')).setValue(plan);
    sh.getRange(uRow, col('planUntil')).setNumberFormat('@').setValue(until.toISOString());
  } finally { lock.releaseLock(); }
}

function payCheck_(req) {
  const auth = auth_(req.token);
  if (!auth) return { ok: false, error: 'auth' };
  const sh = sheet_('Payments', PAY_HEADERS), last = sh.getLastRow();
  if (last < 2) return { ok: true, state: 'none' };
  const rows = sh.getRange(2, 1, last - 1, 2).getValues();
  let found = 0;
  for (let i = rows.length - 1; i >= 0; i--) {
    if (String(rows[i][1]) === String(auth.user.id) && (!req.bill || String(rows[i][0]) === String(req.bill))) { found = i + 2; break; }
  }
  if (!found) return { ok: true, state: 'none' };
  const p = payVerify_(found);
  return { ok: true, state: p.state, plan: p.plan, user: publicUser_(readUser_(auth.row)) };
}

/** Kirishda: oxirgi 3 kundagi kutilayotgan to'lovlarni tekshiradi (daqiqasiga 1 marta) */
function payPendingFor_(userId) {
  if (!tcKey_()) return false;
  const cache = CacheService.getScriptCache(), ck = 'pp_' + userId;
  if (cache.get(ck)) return false;
  cache.put(ck, '1', 60);
  const sh = sheet_('Payments', PAY_HEADERS), last = sh.getLastRow();
  if (last < 2) return false;
  const vals = sh.getRange(2, 1, last - 1, PAY_HEADERS.length).getValues();
  let changed = false, n = 0;
  for (let i = vals.length - 1; i >= 0 && n < 3; i--) {
    if (String(vals[i][1]) !== String(userId) || vals[i][5] !== 'pending') continue;
    if (Date.now() - new Date(vals[i][6]).getTime() > 3 * 86400000) continue;
    n++;
    if (payVerify_(i + 2).state === 'paid') changed = true;
  }
  return changed;
}

  let total = 0;
  const list = sh.getRange(Math.max(2, last - 299), 1, Math.min(300, last - 1), PAY_HEADERS.length).getValues().reverse().map(function (r) {

}

function mcp_(e) {
  let body = null;
  try { body = JSON.parse((e.postData && e.postData.contents) || 'null'); } catch (x) {}
  const who = mcpWho_(String(e.parameter.mcp || ''));
  if (!who) return json_({ jsonrpc: '2.0', id: body && body.id != null ? body.id : null, error: { code: -32001, message: 'Kalit noto\'g\'ri yoki eskirgan. Gapir AI kabinetidan yangi manzil oling.' } });
  if (!body) return json_({ jsonrpc: '2.0', id: null, error: { code: -32700, message: 'Parse error' } });
  MCP_WHO = who;
  if (Array.isArray(body)) {
    const out = body.map(mcpOne_).filter(function (x) { return x; });
    return out.length ? json_(out) : ContentService.createTextOutput('');
  }
  const r = mcpOne_(body);
  return r ? json_(r) : ContentService.createTextOutput('');
}

function mcpOne_(m) {
  if (!m || m.id == null) return null;                  // bildirishnomalar (notifications) — javob kerak emas
  const ok = function (result) { return { jsonrpc: '2.0', id: m.id, result: result }; };
  const params = m.params || {};
  try {
    switch (m.method) {
      case 'initialize':
        return ok({ protocolVersion: params.protocolVersion || '2025-06-18', capabilities: { tools: {} },
          serverInfo: { name: 'gapir-ai', title: 'Gapir AI', version: '1.0.0' },
          instructions: "Gapir AI — o'zbekcha matnni ovozga aylantirish. Avval list_voices bilan ovoz tanlang, keyin generate_speech bilan audio yarating. Natijadagi raw_url dan WAV faylni yuklab oling." });
      case 'ping': return ok({});
      case 'tools/list': return ok({ tools: MCP_TOOLS });
      case 'tools/call': return ok(mcpCall_(params.name, params.arguments || {}));
      case 'resources/list': return ok({ resources: [] });
      case 'prompts/list': return ok({ prompts: [] });
      default: return { jsonrpc: '2.0', id: m.id, error: { code: -32601, message: 'Method not found: ' + m.method } };
    }
  } catch (err) {
    return ok({ isError: true, content: [{ type: 'text', text: 'Xato: ' + (err && err.message || err) }] });
  }
}

/** Gemini qaytargan PCM (L16) ni WAV ga o'giradi */
function toWav_(b64, mime) {
  const pcm = Utilities.base64Decode(b64);
  if (/wav/i.test(mime || '')) return { b64: b64, seconds: (pcm.length - 44) / 48000 };
  const rate = Number((/rate=(\d+)/.exec(mime || '') || [])[1]) || 24000;
  const n = pcm.length, h = [];
  const s = function (str) { for (let i = 0; i < str.length; i++) h.push(str.charCodeAt(i)); };
  const u32 = function (v) { h.push(v & 255, (v >> 8) & 255, (v >> 16) & 255, (v >> 24) & 255); };
  const u16 = function (v) { h.push(v & 255, (v >> 8) & 255); };
  s('RIFF'); u32(36 + n); s('WAVE'); s('fmt '); u32(16); u16(1); u16(1); u32(rate); u32(rate * 2); u16(2); u16(16); s('data'); u32(n);
  const head = h.map(function (b) { return b > 127 ? b - 256 : b; });
  return { b64: Utilities.base64Encode(head.concat(Array.isArray(pcm) ? pcm : Array.prototype.slice.call(pcm))), seconds: n / 2 / rate };
}

/** GitHub'dagi 3 kundan eski audiolarni o'chiradi (bir chaqiruvda 8 tagacha) */
function mcpCleanup_() {
  const H = { Authorization: 'Bearer ' + ghToken_(), Accept: 'application/vnd.github+json' };
  const r = UrlFetchApp.fetch('https://api.github.com/repos/' + ghRepo_() + '/contents/mcp-audio', { headers: H, muteHttpExceptions: true });
  if (r.getResponseCode() !== 200) return;
  const limit = Utilities.formatDate(new Date(Date.now() - 3 * 86400000), CFG.TZ, 'yyyyMMdd');
  JSON.parse(r.getContentText()).filter(function (f) { return f.type === 'file' && f.name.slice(0, 8) < limit; }).slice(0, 8).forEach(function (f) {
    UrlFetchApp.fetch('https://api.github.com/repos/' + ghRepo_() + '/contents/' + f.path, { method: 'delete', headers: H, contentType: 'application/json',
      payload: JSON.stringify({ message: 'Eski audio: ' + f.name, sha: f.sha }), muteHttpExceptions: true });
  });
}

function svcState_(u) {

/** Xizmat limitidan 1 dona band qiladi */
function svcReserve_(row, kind) {
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    const u = readUser_(row), st = svcState_(u)[kind];
    if (!st) return { ok: false, error: "Noma'lum xizmat" };
    if (st.limit <= 0) return { ok: false, error: SVC_NAMES[kind].charAt(0).toUpperCase() + SVC_NAMES[kind].slice(1) + " sizning tarifingizda yo'q. Tarifni oshiring.", upgrade: true };
    if (st.used >= st.limit) return { ok: false, error: 'Bugungi ' + SVC_NAMES[kind] + ' limiti tugadi (' + st.used + '/' + st.limit + '). Ertaga yangilanadi yoki tarifni oshiring.', upgrade: true };
    let j = {}; try { j = JSON.parse(String(u.svc || '{}')); } catch (e) {}
    if (j.day !== today_()) j = { day: today_() };
    j[kind] = (Number(j[kind]) || 0) + 1;
    sheet_('Users', USER_HEADERS).getRange(row, USER_HEADERS.indexOf('svc') + 1).setNumberFormat('@').setValue(JSON.stringify(j));
    return { ok: true, used: j[kind], limit: st.limit };
  } finally { lock.releaseLock(); }
}
function svcRefund_(row, kind) {
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    const u = readUser_(row); let j = {}; try { j = JSON.parse(String(u.svc || '{}')); } catch (e) {}
    if (j.day !== today_() || !j[kind]) return;
    j[kind] = Math.max(0, Number(j[kind]) - 1);
    sheet_('Users', USER_HEADERS).getRange(row, USER_HEADERS.indexOf('svc') + 1).setValue(JSON.stringify(j));
  } finally { lock.releaseLock(); }
}

function geminiErr_(code, raw) {
  if (code === 429 && /limit: ?0|free_tier|billing/i.test(raw)) return "Bu xizmat uchun Google'da to'lov (billing) yoqilmagan. Admin bilan bog'laning.";
  if (code === 429 || code === 503) return "Hozir so'rovlar ko'p. Bir daqiqadan keyin urinib ko'ring.";
  if (code === 400 && /safety|blocked|prohibit/i.test(raw)) return "Bu so'rov xavfsizlik qoidalariga to'g'ri kelmadi. Boshqacha yozib ko'ring.";
  if (code === 401 || code === 403) return "Server API kaliti bilan muammo. Admin bilan bog'laning.";
  if (code === 404) return "Model topilmadi. Admin bilan bog'laning.";
  return "Xatolik yuz berdi (" + code + "). Qayta urinib ko'ring.";
}

  if (!prompt) return { ok: false, error: "Nima chizish yoki o'zgartirishni yozing" };

    const model = CFG.IMAGE_MODELS[i];
    const gen = { responseModalities: ['TEXT', 'IMAGE'] };
    if (aspect && !imgs.length) gen.imageConfig = { aspectRatio: aspect };
    const resp = UrlFetchApp.fetch('https://generativelanguage.googleapis.com/v1beta/models/' + model + ':generateContent', {
      method: 'post', contentType: 'application/json', headers: { 'x-goog-api-key': key }, muteHttpExceptions: true,
      payload: JSON.stringify({ contents: [{ role: 'user', parts: parts }], generationConfig: gen })
    });
    code = resp.getResponseCode(); raw = resp.getContentText();
    if (code === 200) {
      try {
        const ps = ((((JSON.parse(raw).candidates || [])[0] || {}).content || {}).parts) || [];
        const im = ps.filter(function (p) { return (p.inlineData || p.inline_data) && !p.thought; }).pop();

  }
  svcRefund_(auth.row, 'img');
  log_(auth.user.id, 0, 'image', false, 'img: ' + lastErr);
  return { ok: false, error: code === 200 ? "Rasm chiqmadi — so'rovni aniqroq yozib ko'ring." : geminiErr_(code, raw) };

function b64url_(bytes) { return Utilities.base64EncodeWebSafe(bytes).replace(/=+$/, ''); }
function signTicket_(obj) {
  const body = b64url_(Utilities.newBlob(JSON.stringify(obj)).getBytes());
  const sig = b64url_(Utilities.computeHmacSha256Signature(body, gapirSecret_()));
  return body + '.' + sig;
}
function readTicket_(t) {
  const parts = String(t || '').split('.');
  if (parts.length !== 2) return null;
  if (b64url_(Utilities.computeHmacSha256Signature(parts[0], gapirSecret_())) !== parts[1]) return null;
  try { return JSON.parse(Utilities.newBlob(Utilities.base64DecodeWebSafe(parts[0] + '==='.slice((parts[0].length + 3) % 4))).getDataAsString()); } catch (e) { return null; }
}
function ticket_(req) {
  const auth = auth_(req.token);
  if (!auth) return { ok: false, error: 'auth' };
  const kind = req.kind === 'vid' ? 'vid' : 'site';
  const S = set_();
  if (!S.mcpBase) return { ok: false, error: "Xizmat hali sozlanmagan (Worker manzili yo'q). Admin bilan bog'laning." };
  const r = svcReserve_(auth.row, kind);
  if (!r.ok) return r;
  const n = Utilities.getUuid();
  CacheService.getScriptCache().put('tk_' + n, 'new', 1800);
  log_(auth.user.id, 0, kind === 'vid' ? CFG.VIDEO_MODEL : 'site', true, kind + ' start');
  return { ok: true, worker: S.mcpBase, ticket: signTicket_({ u: auth.user.id, k: kind, n: n, e: Date.now() + 20 * 60000 }), used: r.used, limit: r.limit };
}
/** Worker chiptani ishlatadi: bir chipta — bitta ish */
function ticketUse_(req) {
  const t = readTicket_(req.ticket);
  if (!t || t.e < Date.now()) return { ok: false, error: "Chipta eskirgan. Sahifani yangilab qayta urinib ko'ring." };
  const c = CacheService.getScriptCache();
  if (c.get('tk_' + t.n) !== 'new') return { ok: false, error: 'Chipta allaqachon ishlatilgan' };
  c.put('tk_' + t.n, 'used', 1800);
  return { ok: true, kind: t.k, user: t.u };
}
/** Worker ish bajara olmasa — limitni qaytaradi (bir chipta bir marta) */
function svcRefundTicket_(req) {
  const t = readTicket_(req.ticket);
  if (!t) return { ok: false, error: 'bad ticket' };
  const c = CacheService.getScriptCache();
  if (c.get('tk_' + t.n) !== 'used') return { ok: true, already: true };
  c.put('tk_' + t.n, 'done', 1800);
  const row = findRow_('Users', USER_HEADERS, 'id', String(t.u));
  if (row) svcRefund_(row, t.k);
  log_(String(t.u), 0, t.k === 'vid' ? CFG.VIDEO_MODEL : 'site', false, t.k + ': ' + String(req.reason || '').slice(0, 200));
  return { ok: true };
}

/* ---- GitHub: sayt joylash va APK yig'ish ---- */
function ghPut_(path, content, msg) {
  const H = { Authorization: 'Bearer ' + ghToken_(), Accept: 'application/vnd.github+json' };
  const url = 'https://api.github.com/repos/' + ghRepo_() + '/contents/' + path;
  let sha = null;
  const g = UrlFetchApp.fetch(url, { headers: H, muteHttpExceptions: true });
  if (g.getResponseCode() === 200) sha = JSON.parse(g.getContentText()).sha;
  const body = { message: msg, content: Utilities.base64Encode(content, Utilities.Charset.UTF_8) };
  if (sha) body.sha = sha;
  const r = UrlFetchApp.fetch(url, { method: 'put', headers: H, contentType: 'application/json', payload: JSON.stringify(body), muteHttpExceptions: true });
  return r.getResponseCode() < 300;
}
function slug_(s) {
  const t = String(s || '').toLowerCase().replace(/[ʻʼ‘’'`]/g, '').replace(/[^a-z0-9]+/g, '-').replace(/^-+|-+$/g, '').slice(0, 30);
  return t || 'sayt';
}
function cleanHtml_(h) {
  h = String(h || '');
  if (h.length < 50 || h.length > 1500000 || !/<html|<body|<div/i.test(h)) return '';
  return h;
}
function sitePublish_(req) {
  const auth = auth_(req.token);
  if (!auth) return { ok: false, error: 'auth' };
  if (!ghToken_()) return { ok: false, error: "Joylash sozlanmagan (GITHUB_TOKEN yo'q). Admin bilan bog'laning." };
  const html = cleanHtml_(req.html);
  if (!html) return { ok: false, error: 'Sayt kodi topilmadi' };
  const id = slug_(req.name) + '-' + String(auth.user.id).slice(0, 4) + Utilities.getUuid().slice(0, 4);
  const name = String(req.site || '').replace(/[^a-z0-9-]/g, '').slice(0, 50) || id;
  if (!ghPut_('sites/' + name + '/index.html', html, 'Gapir AI sayt: ' + name)) return { ok: false, error: "GitHub'ga yozib bo'lmadi" };
  log_(auth.user.id, 0, 'github', true, 'site publish ' + name);
  return { ok: true, site: name, url: CFG.SITE_URL + 'sites/' + name + '/' };
}
function appBuild_(req) {
  const auth = auth_(req.token);
  if (!auth) return { ok: false, error: 'auth' };
  if (!ghToken_()) return { ok: false, error: "APK yig'ish sozlanmagan (GITHUB_TOKEN yo'q). Admin bilan bog'laning." };
  const html = cleanHtml_(req.html);
  if (!html) return { ok: false, error: 'Ilova kodi topilmadi' };
  const r = svcReserve_(auth.row, 'app');
  if (!r.ok) return r;
  const name = String(req.name || 'Mening ilovam').replace(/[<>"&\\]/g, '').trim().slice(0, 30) || 'Mening ilovam';
  const id = 'a' + Utilities.formatDate(new Date(), CFG.TZ, 'yyMMddHHmmss') + Utilities.getUuid().slice(0, 4);
  const okHtml = ghPut_('apps/' + id + '/index.html', html, 'Gapir AI ilova: ' + id);
  const okMeta = okHtml && ghPut_('apps/' + id + '/meta.json', JSON.stringify({ name: name, user: auth.user.id, created: new Date().toISOString() }), 'meta ' + id);
  if (!okMeta) { svcRefund_(auth.row, 'app'); return { ok: false, error: "GitHub'ga yozib bo'lmadi" }; }
  // meta.json yozilishi GitHub Actions'dagi "Build APK" ishini avtomatik boshlaydi
  log_(auth.user.id, 0, 'github', true, 'app build ' + id);
  return { ok: true, id: id, used: r.used, limit: r.limit };
}
function appStatus_(req) {
  const auth = auth_(req.token);
  if (!auth) return { ok: false, error: 'auth' };
  const id = String(req.id || '').replace(/[^a-z0-9]/g, '');
  if (!id) return { ok: false, error: 'id' };
  const H = { Authorization: 'Bearer ' + ghToken_(), Accept: 'application/vnd.github+json' };
  const r = UrlFetchApp.fetch('https://api.github.com/repos/' + ghRepo_() + '/releases/tags/app-' + id, { headers: H, muteHttpExceptions: true });
  if (r.getResponseCode() === 200) {
    const a = (JSON.parse(r.getContentText()).assets || []).filter(function (x) { return /\.apk$/.test(x.name); })[0];
    if (a) return { ok: true, state: 'done', url: a.browser_download_url, size: a.size };
  }
  // yig'ish xato bilan tugaganmi?
  const runs = UrlFetchApp.fetch('https://api.github.com/repos/' + ghRepo_() + '/actions/runs?event=repository_dispatch&per_page=10', { headers: H, muteHttpExceptions: true });
  if (runs.getResponseCode() === 200) {
    const run = (JSON.parse(runs.getContentText()).workflow_runs || []).filter(function (w) { return String(w.display_title || w.name || '').indexOf(id) >= 0; })[0];
    if (run && run.status === 'completed' && run.conclusion !== 'success') return { ok: true, state: 'failed' };
  }
  return { ok: true, state: 'building' };
}

/** Admin: Google hisobida qaysi xizmatlar ochiqligini tekshiradi */
function adminProbe_() {
  const key = PropertiesService.getScriptProperties().getProperty('GEMINI_API_KEY');
  const out = { ok: true, models: {}, image: null, github: !!ghToken_(), worker: !!set_().mcpBase, secret: gapirSecret_().slice(0, 6) + '…' };
  const lr = UrlFetchApp.fetch('https://generativelanguage.googleapis.com/v1beta/models?pageSize=200', { headers: { 'x-goog-api-key': key }, muteHttpExceptions: true });
  if (lr.getResponseCode() === 200) {
    const names = (JSON.parse(lr.getContentText()).models || []).map(function (m) { return String(m.name).replace('models/', ''); });
    CFG.IMAGE_MODELS.concat([CFG.VIDEO_MODEL]).forEach(function (m) { out.models[m] = names.indexOf(m) >= 0; });
    out.videoModels = names.filter(function (n) { return /omni|veo/.test(n); });
    out.imageModels = names.filter(function (n) { return /image/.test(n); });
  } else out.listError = lr.getResponseCode();
  // haqiqiy kichik rasm sinovi (pullik bo'lsa ~0,04$)
  for (let i = 0; i < CFG.IMAGE_MODELS.length; i++) {
    const m = CFG.IMAGE_MODELS[i];
    if (out.imageModels && out.imageModels.length && out.models[m] === false) continue;
    const r = UrlFetchApp.fetch('https://generativelanguage.googleapis.com/v1beta/models/' + m + ':generateContent', {
      method: 'post', contentType: 'application/json', headers: { 'x-goog-api-key': key }, muteHttpExceptions: true,
      payload: JSON.stringify({ contents: [{ parts: [{ text: 'A small plain blue circle on white background' }] }], generationConfig: { responseModalities: ['TEXT', 'IMAGE'] } })
    });
    const c = r.getResponseCode();
    if (c === 404) continue;
    out.image = { model: m, code: c, ok: c === 200, message: c === 200 ? 'Ishlayapti' : geminiErr_(c, r.getContentText()), raw: c === 200 ? '' : r.getContentText().slice(0, 300) };
    break;
  }
  return out;
}
